# Optional libraries. PythonOS uses each one when it is installed and works without it, so a failure to install any of them is never a problem.
# (requirements.txt is the list PythonOS needs to start; changing it forces a new package on Android and the ISO, so it is kept short.)
python-dateutil     # date -d "next friday", and any date the built-in parser does not know
humanize            # "3 minutes ago" in stat, uptime and others
zxcvbn              # honest advice on how strong a new password is
filetype            # file: what a file is, from its first bytes (and cat does not print a picture as text)
beautifulsoup4      # web: readable text from a page, with its links
pyyaml              # convert: YAML to JSON and back
rapidfuzz           # better "did you mean" and help search
feedparser          # rss: feeds that are not quite valid
charset-normalizer  # cat, less, head, tail, grep, diff, wc: read text files in Windows-1252, UTF-16, Shift-JIS and so on, not only UTF-8
py7zr; python_version < "3.14"       # zip, unzip: .7z archives
distro              # sysinfo: the name of the Linux distribution
py-cpuinfo          # lscpu, sysinfo: the processor's model on Windows and Android
watchfiles; python_version < "3.14"  # watch -f: run a command again when a file or folder changes (without it the folder is checked once a second)
dnspython           # nslookup: more record types (SRV, CAA, DS, DNSKEY, TLSA, ...)
